# Test için yönetici 2026-12-13 tarihine Mühendislik yemekhanesinde kahvaltı ve öğle menüsü açar
$ curl -s -H 'Cookie: oturum=3b4158cf…' 'http://localhost:8080/api/menu?yemekhane_id=2&tarih=2026-12-13&ogun_id=2'   (ve ogun_id=1)
# Öğle menüsü 86 · öğrenci fiyatı: 40 TL, yarım porsiyon: 24 TL (oran 0.6)
# Kahvaltı menüsü 85 · yarım porsiyon fiyatı: null (kahvaltıda yarım porsiyon yok)
# Kahvaltıya yarım porsiyon → yordam reddeder:
$ curl -s -H 'Cookie: oturum=3b4158cf…' -X POST -H Content-Type: application/json -d {"menu_id":85,"porsiyon":"yarim"} http://localhost:8080/api/rezervasyonlar
{"hata":"Bu öğünde yarım porsiyon seçilemez."}
# Geçersiz porsiyon değeri → reddedilir:
$ curl -s -H 'Cookie: oturum=3b4158cf…' -X POST -H Content-Type: application/json -d {"menu_id":86,"porsiyon":"cift"} http://localhost:8080/api/rezervasyonlar
{"hata":"Geçersiz porsiyon seçimi."}
# Öğleye yarım porsiyon → ücret tam fiyatın %60'ı:
$ curl -s -H 'Cookie: oturum=3b4158cf…' -X POST -H Content-Type: application/json -d {"menu_id":86,"porsiyon":"yarim"} http://localhost:8080/api/rezervasyonlar
{"rezervasyon_id":368,"ucret":24,"bakiye":331,"porsiyon":"yarim"}
# Porsiyon gönderilmezse tam porsiyon (kahvaltı):
